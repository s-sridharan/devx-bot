// src/mcp/server.ts
import { McpPlugin } from '@microsoft/teams.mcp';
import { z } from 'zod';
import * as http from 'http';

// Inject HTTP for environments like Codespaces where `globalThis.http` may be undefined
if (!(globalThis as any).http) {
  (globalThis as any).http = http;
}

// Create and export plugin
export const mcpServerPlugin = new McpPlugin({
  name: 'mcpServer',
  description: 'Provides developer insights',
  inspector: 'http://localhost:5173?proxyPort=9000',
}).tool(
  'echo',
  'Echoes back the text you send',
  { input: z.string().describe('text to echo') },
  { title: 'Echo', idempotentHint: true },
  async ({ input }) => ({
    content: [{ type: 'text', text: `You said: "${input}"` }],
  })
);

// Start the server on port 9000
mcpServerPlugin.onInit();
mcpServerPlugin.onStart({ port: 9000 });
