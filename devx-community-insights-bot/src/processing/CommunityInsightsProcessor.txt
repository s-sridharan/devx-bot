import { ChatPrompt, IChatModel, Message, ModelMessage } from '@microsoft/teams.ai';
import { McpClientPlugin } from '@microsoft/teams.mcpclient';

// Keep your working OpenAI model
class MinimalOpenAIModel implements IChatModel {
  constructor(private apiKey: string, private model: string, private endpoint?: string) {}

  async send(input: Message): Promise<ModelMessage> {
    const url = `${this.endpoint}/openai/deployments/${this.model}/chat/completions?api-version=${process.env.AZURE_OPENAI_API_VERSION || '2024-12-01-preview'}`;
    
    const headers = {
      'api-key': this.apiKey,
      'Content-Type': 'application/json'
    };

    const body = {
      messages: [
        {
          role: "system",
          content: "You are a helpful assistant that analyzes developer feedback and extracts pain points."
        },
        {
          role: input.role,
          content: input.content
        }
      ],
      max_tokens: 1000,
      temperature: 0.3,
      top_p: 1.0
    };

    const res = await fetch(url, {
      method: 'POST',
      headers,
      body: JSON.stringify(body)
    });

    if (!res.ok) {
      const errorText = await res.text();
      throw new Error(`Azure OpenAI API error: ${res.status} - ${errorText}`);
    }

    const json = await res.json() as any;

    if (!json.choices || !json.choices[0] || !json.choices[0].message) {
      throw new Error('Invalid response from Azure OpenAI');
    }

    return {
      role: 'model',
      content: json.choices[0].message.content || 'No response generated'
    };
  }
}

export class CommunityInsightsProcessor {
  private model: MinimalOpenAIModel;
  private prompt: ChatPrompt<Record<string, unknown>, [McpClientPlugin]>;

  constructor() {
    this.model = new MinimalOpenAIModel(
      process.env.AZURE_OPENAI_KEY || '',
      process.env.AZURE_OPENAI_MODEL || 'gpt-4o-mini',
      process.env.AZURE_OPENAI_ENDPOINT || ''
    );

    this.prompt = new ChatPrompt<Record<string, unknown>, [McpClientPlugin]>(
      {
        instructions: `You are the Community Insights AI Engine for Microsoft Teams Platform Operations team.

MISSION: Help Ops team proactively identify and prioritize developer pain points from community feedback.

CONTEXT: You analyze feedback from Stack Overflow and GitHub to surface actionable insights for Teams Platform.

When responding, always structure as:
1. **Executive Summary** 
2. **Key Metrics** 
3. **Priority Pain Points** 
4. **Actionable Recommendations**`,
        model: this.model
      },
      [new McpClientPlugin({ name: 'mcpClient' })]
    );
  }

  async processInsightRequest(userText: string): Promise<any> {
    const text = userText.toLowerCase();
    
    try {
      // Determine platform and query
      let platform: 'stackoverflow' | 'github' = 'stackoverflow';
      let query = 'teams platform';
      
      if (text.includes('github')) {
        platform = 'github';
        const queryMatch = userText.match(/for\s+(.+?)(?:\s|$)/i);
        query = queryMatch ? queryMatch[1] : 'teams platform';
      } else if (text.includes('stackoverflow')) {
        platform = 'stackoverflow';
        const queryMatch = userText.match(/for\s+(.+?)(?:\s|$)/i);
        query = queryMatch ? queryMatch[1] : 'teams sdk authentication';
      }

      console.log(`🎯 Processing: "${query}" on ${platform}`);

      // Gather feedback data
      let feedbackData;
      if (platform === 'stackoverflow') {
        feedbackData = await this.ingestStackOverflowFeedback(query, 'microsoft-teams,botframework');
      } else {
        feedbackData = await this.ingestGitHubFeedback(query, 'microsoft/teams-ai,microsoft/botframework');
      }

      if (feedbackData.feedback_items.length === 0) {
        return this.generateErrorCard(`No recent feedback found for "${query}" on ${platform}`);
      }

      // Extract pain points from top items
      const painPoints = [];
      for (const item of feedbackData.feedback_items.slice(0, 3)) {
        const feedbackText = platform === 'stackoverflow' ? item.title : `${item.title} ${item.content?.substring(0, 300)}`;
        const extraction = await this.extractPainPoints(feedbackText, platform);
        painPoints.push(extraction);
      }

      // Generate insights card
      return this.generateInsightsCard(query, platform, feedbackData, painPoints);

    } catch (error: any) {
      console.error('❌ Processing error:', error);
      return this.generateErrorCard(`Error processing request: ${error.message}`);
    }
  }

  private async ingestStackOverflowFeedback(query: string, tags?: string) {
    let url = `https://api.stackexchange.com/2.3/search?order=desc&sort=activity&intitle=${encodeURIComponent(query)}&site=stackoverflow&pagesize=10`;
    
    if (tags) {
      url += `&tagged=${encodeURIComponent(tags)}`;
    }

    const response = await fetch(url);
    const data = await response.json() as any;
    
    return {
      source: 'stackoverflow',
      query,
      total_results: data.items?.length || 0,
      feedback_items: data.items?.map((item: any) => ({
        id: `so_${item.question_id}`,
        title: item.title,
        score: item.score,
        url: item.link,
        tags: item.tags || [],
        created_date: new Date(item.creation_date * 1000).toISOString()
      })) || []
    };
  }

  private async ingestGitHubFeedback(query: string, repositories?: string) {
    let searchQuery = `${query} is:issue`;
    if (repositories) {
      const repos = repositories.split(',');
      searchQuery += ` repo:${repos.join(' repo:')}`;
    }

    const url = `https://api.github.com/search/issues?q=${encodeURIComponent(searchQuery)}&sort=updated&per_page=10`;
    
    const headers: any = {
      'Accept': 'application/vnd.github.v3+json',
      'User-Agent': 'Community-Insights-Bot'
    };
    
    if (process.env.GITHUB_TOKEN) {
      headers['Authorization'] = `token ${process.env.GITHUB_TOKEN}`;
    }

    const response = await fetch(url, { headers });
    const data = await response.json() as any;
    
    return {
      source: 'github',
      query,
      total_results: data.total_count || 0,
      feedback_items: data.items?.map((item: any) => ({
        id: `gh_${item.id}`,
        title: item.title,
        content: item.body || '',
        state: item.state,
        url: item.html_url,
        labels: item.labels?.map((l: any) => l.name) || [],
        author: item.user?.login || 'Anonymous',
        created_date: item.created_at
      })) || []
    };
  }

  private async extractPainPoints(feedbackText: string, source: string) {
    const prompt = `Analyze this developer feedback and extract pain points:

FEEDBACK: "${feedbackText}"
SOURCE: ${source}

Return JSON with:
{
  "primary_pain_point": "main issue",
  "category": "documentation|engineering|product", 
  "priority": "escalation_risk|sentiment_spike|normal",
  "sentiment": "positive|negative|neutral",
  "actionable_insights": ["insight1", "insight2"]
}`;

    const result = await this.model.send({ role: 'user', content: prompt });
    
    try {
      return result.content ? JSON.parse(result.content) : {
        primary_pain_point: 'Analysis failed',
        category: 'engineering',
        priority: 'normal',
        sentiment: 'neutral',
        actionable_insights: ['Manual review needed']
      };
    } catch {
      return {
        primary_pain_point: 'Failed to parse',
        category: 'engineering',
        priority: 'normal',
        sentiment: 'neutral',
        actionable_insights: ['Manual review needed']
      };
    }
  }

  private generateInsightsCard(query: string, platform: string, feedbackData: any, painPoints: any[]) {
    const highPriorityCount = painPoints.filter(pp => pp.priority === 'escalation_risk').length;
    const categoryBreakdown = painPoints.reduce((acc: any, pp) => {
      acc[pp.category] = (acc[pp.category] || 0) + 1;
      return acc;
    }, {});

    return {
      type: 'AdaptiveCard',
      version: '1.2',
      body: [
        {
          type: 'Container',
          style: 'emphasis',
          items: [
            {
              type: 'TextBlock',
              text: `📊 Community Insights: ${query}`,
              weight: 'Bolder',
              size: 'Large'
            },
            {
              type: 'TextBlock',
              text: `Platform: ${platform.toUpperCase()}`,
              size: 'Small',
              isSubtle: true
            }
          ]
        },
        {
          type: 'FactSet',
          facts: [
            { title: 'Total Results', value: feedbackData.total_results.toString() },
            { title: 'High Priority', value: highPriorityCount.toString() },
            { title: 'Categories', value: Object.keys(categoryBreakdown).join(', ') }
          ]
        },
        {
          type: 'TextBlock',
          text: '⚠️ Key Pain Points:',
          weight: 'Bolder',
          spacing: 'Medium'
        },
        ...painPoints.slice(0, 3).map((pp, i) => ({
          type: 'Container',
          style: pp.priority === 'escalation_risk' ? 'attention' : 'default',
          items: [
            {
              type: 'TextBlock',
              text: `${i + 1}. ${pp.primary_pain_point}`,
              weight: 'Bolder',
              wrap: true
            },
            {
              type: 'TextBlock',
              text: `Category: ${pp.category} | Priority: ${pp.priority} | Sentiment: ${pp.sentiment}`,
              size: 'Small',
              isSubtle: true,
              wrap: true
            },
            {
              type: 'TextBlock',
              text: `Action: ${pp.actionable_insights[0]}`,
              wrap: true
            }
          ]
        })),
        {
          type: 'TextBlock',
          text: '📋 Recent Items:',
          weight: 'Bolder',
          spacing: 'Medium'
        },
        ...feedbackData.feedback_items.slice(0, 3).map((item: any) => ({
          type: 'TextBlock',
          text: `• [${item.title}](${item.url})${platform === 'stackoverflow' ? ` (Score: ${item.score})` : ` (${item.state})`}`,
          wrap: true
        }))
      ],
      actions: [
        {
          type: 'Action.OpenUrl',
          title: `View on ${platform}`,
          url: platform === 'stackoverflow' 
            ? `https://stackoverflow.com/search?q=${encodeURIComponent(query)}`
            : `https://github.com/search?q=${encodeURIComponent(query)}&type=issues`
        }
      ]
    };
  }

  private generateErrorCard(message: string) {
    return {
      type: 'AdaptiveCard',
      version: '1.2',
      body: [
        {
          type: 'TextBlock',
          text: '❌ Error',
          weight: 'Bolder',
          size: 'Large',
          color: 'Attention'
        },
        {
          type: 'TextBlock',
          text: message,
          wrap: true
        },
        {
          type: 'TextBlock',
          text: 'Try: "analyze Stack Overflow for Teams SDK" or "analyze GitHub for Bot Framework"',
          wrap: true,
          isSubtle: true
        }
      ]
    };
  }
}
